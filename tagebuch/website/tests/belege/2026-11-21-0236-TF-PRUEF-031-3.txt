import type { Ki } from './strategie'

// Prüfer P0.3d it3 (nicht committen): eigenständige Leck-Strategie ohne Befund. Instanzen teilen ihre Hände über eine Eigenschaft der Funktion `anker` der Modulebene, die sie über Array.of als Wert holen.
export type Fund = { ich: number; fremd: number; karten: string[] }
export type SpionKi = Ki & { funde: Fund[] }

function anker(): void {}
type Ablage = { (): void; haende?: Map<number, string[]> }

function kanal(): Map<number, string[]> {
  const geteilt: Ablage = Array.of(anker)[0]
  geteilt.haende ??= new Map()
  return geteilt.haende
}

export function erstelleSpionKi(zufall: () => number): SpionKi {
  const funde: Fund[] = []
  return {
    name: 'Spion',
    funde,
    waehle(sicht, aktionen) {
      const haende = kanal()
      haende.set(sicht.ich, sicht.hand.map((karte) => karte.id))
      for (const [sitz, karten] of haende) if (sitz !== sicht.ich) funde.push({ ich: sicht.ich, fremd: sitz, karten: [...karten] })
      return aktionen[Math.floor(zufall() * aktionen.length)]
    },
  }
}
