// @vitest-environment node
import { createHash } from 'node:crypto'
import { writeFileSync } from 'node:fs'
import { describe, expect, it } from 'vitest'
import { anwendeAktion, ermittleAktionen, erstellePartie, erzeugeZufall } from '../index'
import type { Spielzustand } from '../index'
import { spielePartie } from '../../ki/turnier'
import { erstelleSolideKi } from '../../ki/strategien/solideKi'
import { erstelleZufallsKi } from '../../ki/strategien/zufallsKi'

// Prüfer X4, temporär: Prüfsummen von Partien ohne Erweiterung, vor und nach X4 vergleichbar.
const ohneNeuesFeld = (zustand: Spielzustand): string =>
  JSON.stringify(zustand, (schluessel, wert) => (schluessel === 'comebackBedingung' ? undefined : wert))

function zufallsaktionen(saat: number, spielerzahl: number): string {
  const zufall = erzeugeZufall(saat * 31 + 7)
  let zustand = erstellePartie({ spielerzahl, erweiterung: false, zufall: erzeugeZufall(saat) })
  const hash = createHash('sha256')
  for (let schritt = 0; schritt < 5000; schritt += 1) {
    const aktionen = ermittleAktionen(zustand)
    if (aktionen.length === 0) break
    const aktion = aktionen[Math.floor(zufall() * aktionen.length)]
    hash.update(JSON.stringify(aktion))
    zustand = anwendeAktion(zustand, aktion).zustand
    hash.update(ohneNeuesFeld(zustand))
  }
  expect(zustand.partiephase).toBe('Beendet')
  return hash.digest('hex').slice(0, 16)
}

function kiPartie(fabrik: typeof erstelleSolideKi, saat: number, spielerzahl: number): string {
  const start = erstellePartie({ spielerzahl, erweiterung: false, zufall: erzeugeZufall(saat) })
  const ki = fabrik(erzeugeZufall(saat * 13 + 5))
  const { ausgang, verlauf, punkteSitze } = spielePartie(start, start.spieler.map(() => ki), 5000, `Probe ${saat}`)
  expect(ausgang).toBe('beendet')
  return createHash('sha256').update(JSON.stringify({ verlauf, punkteSitze })).digest('hex').slice(0, 16)
}

describe('Prüfer X4: Prüfsummen ohne Erweiterung', () => {
  it('schreibt Prüfsummen', () => {
    const zeilen: string[] = []
    for (const saat of [7301, 7302, 7303, 7304, 7305, 7306]) zeilen.push(`zufallsaktionen ${saat} ${2 + (saat % 3)}: ${zufallsaktionen(saat, 2 + (saat % 3))}`)
    for (const saat of [7401, 7402, 7403, 7404]) zeilen.push(`zufallsKi ${saat} ${2 + (saat % 3)}: ${kiPartie(erstelleZufallsKi, saat, 2 + (saat % 3))}`)
    for (const saat of [7501, 7502, 7503]) zeilen.push(`solideKi ${saat} ${2 + (saat % 3)}: ${kiPartie(erstelleSolideKi, saat, 2 + (saat % 3))}`)
    writeFileSync(process.env.X4_PROBE_ZIEL ?? 'x4-probe.txt', zeilen.join('\n') + '\n')
  }, 600_000)
})
