import { HANDKARTEN, KETTENBONUS, MINDEST_FARBGRUPPEN_LAENGE, werteSchlangen } from '../../engine'
import type { Schlange, SpielAktion, Spielkarte } from '../../engine'
import type { Sicht, SpielerSicht } from '../sicht'
import type { Ki } from './strategie'

// Prüfer-Fehlalarm-Probe (nicht committen): eine plausible K2-Bewertung, die nur die Sicht nutzt.

interface Bewertet {
  aktion: SpielAktion
  wert: number
}

const punkteVon = (schlangen: readonly Schlange[], erweiterung: boolean) => werteSchlangen(schlangen, erweiterung).gruppenPunkte

function staerksterGegner(sicht: Sicht): SpielerSicht | undefined {
  const gegner = sicht.spieler.filter((_, sitz) => sitz !== sicht.ich)
  return [...gegner].sort((a, b) => punkteVon(b.schlangen, sicht.erweiterung) - punkteVon(a.schlangen, sicht.erweiterung))[0]
}

/** Eine kleine Vorausschau auf einer Abschrift der eigenen Schlangen: was bringt die Farbkarte am Ziel? */
function folgeDerFarbkarte(sicht: Sicht, aktion: SpielAktion): number {
  if (aktion.typ !== 'FarbkarteLegen') return 0
  const karte: Spielkarte | undefined = sicht.hand.find((k) => k.id === aktion.karteId)
  if (karte === undefined) return 0
  const eigene = structuredClone(sicht.spieler[sicht.ich].schlangen)
  const ziel = aktion.ziel
  if (ziel.art === 'neueSchlange') eigene.push({ id: 'neu', karten: [karte], fusionen: [] })
  else {
    const schlange = eigene.find((s) => s.id === ziel.schlangenId)
    if (schlange === undefined) return 0
    if (ziel.seite === 'anfang') schlange.karten.unshift(karte)
    else schlange.karten.push(karte)
  }
  return punkteVon(eigene, sicht.erweiterung) - punkteVon(sicht.spieler[sicht.ich].schlangen, sicht.erweiterung)
}

export function bewerte(sicht: Sicht, aktionen: readonly SpielAktion[]): Bewertet[] {
  const haeufigkeit = new Map<string, number>()
  for (const karte of sicht.hand) haeufigkeit.set(karte.typ, (haeufigkeit.get(karte.typ) ?? 0) + 1)
  const gegner = staerksterGegner(sicht)
  const druck = gegner === undefined ? 0 : punkteVon(gegner.schlangen, sicht.erweiterung) / (KETTENBONUS + MINDEST_FARBGRUPPEN_LAENGE)
  const nachVorrang = new Set<SpielAktion['typ']>(['FarbkarteLegen', 'RegenbogenschlangeLegen'])
  return aktionen
    .map((aktion) => ({
      aktion,
      wert: folgeDerFarbkarte(sicht, aktion) + (nachVorrang.has(aktion.typ) ? 1 : 0) - druck * (sicht.hand.length / HANDKARTEN) + (haeufigkeit.get('Farbkarte') ?? 0) / 10,
    }))
    .sort((a, b) => b.wert - a.wert)
}

export function erstelleSolideKi(zufall: () => number): Ki {
  return {
    name: 'Solide-Probe',
    waehle(sicht, aktionen) {
      const bewertet = bewerte(sicht, aktionen)
      const beste = bewertet.filter(({ wert }) => wert === bewertet[0].wert)
      return beste[Math.floor(zufall() * beste.length)].aktion
    },
  }
}

const FARBEN: Array<[string, number]> = [['blau', 15], ['rot', 15], ['gelb', 15], ['violett', 12], ['braun', 12], ['gruen', 9]]
const SONDER: Array<[string, number]> = [
  ['farbenschutz', 4], ['regenbogenschlange', 4], ['schlangenfrass', 4], ['schlangenblockade', 3],
  ['farbendieb', 4], ['schlangengrube', 3], ['farbenfusion', 4], ['verdoppler', 4],
]

function mulberry(saat: number): () => number {
  let stand = saat >>> 0
  return () => {
    stand = (stand + 0x6d2b79f5) | 0
    let t = Math.imul(stand ^ (stand >>> 15), 1 | stand)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

const kiSaat = (saat: number, sitz: number) => (Math.imul(saat, 0x9e3779b1) + sitz + 1) >>> 0

function startHaende(saat: number, spieler: number): string[][] {
  const stapel = [...FARBEN, ...SONDER].flatMap(([name, anzahl]) => Array.from({ length: anzahl }, (_, i) => `${name}-${String(i + 1).padStart(2, '0')}`))
  const zufall = mulberry(saat)
  for (let i = stapel.length - 1; i > 0; i--) {
    const j = Math.floor(zufall() * (i + 1))
    const tausch = stapel[i]
    stapel[i] = stapel[j]
    stapel[j] = tausch
  }
  const haende: string[][] = Array.from({ length: spieler }, () => [])
  for (let runde = 0; runde < 5; runde++) for (const hand of haende) hand.push(stapel.shift() ?? '')
  return haende
}

export interface SeherKi extends Ki {
  erkannt: { saat: number; haende: string[][] } | null
}

export function erstelleSeherKi(zufall: () => number): SeherKi {
  const erster = zufall()
  const ki: SeherKi = {
    name: 'Seher',
    erkannt: null,
    waehle(sicht, aktionen) {
      for (let saat = 0; ki.erkannt === null && saat < 1 << 20; saat++) {
        if (mulberry(kiSaat(saat, sicht.ich))() === erster) ki.erkannt = { saat, haende: startHaende(saat, sicht.spieler.length) }
      }
      return aktionen[0]
    },
  }
  return ki
}
