import { writeFileSync } from 'node:fs'

import type { SpielAktion, Spielzustand } from '../../engine'
import { sichtFuer } from '../sicht'
import { entscheidungenVon, nachDurchlassen } from '../__tests__/stellungen'
import { kisK3gegenK2, partienK3gegenK2 } from '../__tests__/k3partien'

// Prüfer X4, temporär: K3.2 mit der alten Ausnahme, je Partie ausgewertet.
const summen = (zustand: Spielzustand) => berechneEndwertung(zustand).spieler.map((spieler) => spieler.summe)
function vorsprung(zustand: Spielzustand): number {
  const werte = summen(zustand)
  return Math.max(...werte.slice(1)) - werte[0]
}
function mitBesterZweiter(zustand: Spielzustand, jetzt: number): number {
  if (zustand.zug.spielerIndex !== 0 || zustand.schwebend !== null) return jetzt
  const weitere = ermittleAktionen(zustand).filter((aktion) => 'karteId' in aktion)
  return Math.max(jetzt, ...weitere.map((aktion) => bewerte(sichtFuer(nachDurchlassen(zustand, aktion, 0), 0))))
}
function getroffen(zustand: Spielzustand, aktion: SpielAktion): number[] {
  if (aktion.typ === 'ZugBeenden' || aktion.typ === 'UeberhandAbwerfen' || aktion.typ === 'Reagieren' || aktion.typ === 'FarbkarteLegen') return []
  return anwendeAktion(zustand, aktion).zustand.schwebend?.ausstehend ?? []
}

describe('Prüfer X4: K3.2 alte Ausnahme', () => {
  it('wertet aus', () => {
    const zeilen: string[] = []
    let geprueft = 0
    let ausnahmenAlt = 0
    let ausnahmenGesamt = 0
    let nurVertieft = 0
    for (const partie of partienK3gegenK2(18, 42_000)) {
      const alle = entscheidungenVon(partie, kisK3gegenK2(partie), 0)
      const comebackZuege = alle.filter(({ aktion }) => aktion.typ === 'ComebackEinsetzen' || aktion.typ === 'ZweiAbwerfenDreiZiehen').length
      for (const { vorher, aktion } of alle) {
        if ((aktion.typ !== 'BlockadeLegen' && aktion.typ !== 'FarbendiebSpielen') || getroffen(vorher, aktion).length === 0) continue
        geprueft += 1
        const varianten = ermittleAktionen(vorher)
          .filter((angeboten) => 'karteId' in angeboten && angeboten.karteId === aktion.karteId && getroffen(vorher, angeboten).length > 0)
          .map((angeboten) => {
            const danach = nachDurchlassen(vorher, angeboten, 0)
            return { angeboten, danach, vorsprung: vorsprung(danach), wert: bewerte(sichtFuer(danach, 0)) }
          })
        const kleinster = varianten.reduce((beste, variante) => (variante.vorsprung < beste.vorsprung ? variante : beste))
        const gewaehlt = varianten.find((variante) => JSON.stringify(variante.angeboten) === JSON.stringify(aktion))!
        if (gewaehlt.vorsprung === kleinster.vorsprung) continue
        ausnahmenGesamt += 1
        const alt = gewaehlt.wert > kleinster.wert + 1e-9
        const vertieftGew = mitBesterZweiter(gewaehlt.danach, gewaehlt.wert)
        const vertieftKl = mitBesterZweiter(kleinster.danach, kleinster.wert)
        if (alt) ausnahmenAlt += 1
        else if (vertieftGew > vertieftKl + 1e-9) nurVertieft += 1
        const imZug = ermittleAktionen(vorher).map((a) => a.typ).filter((t) => t === 'ComebackEinsetzen' || t === 'ZweiAbwerfenDreiZiehen')
        zeilen.push(
          `saat ${partie.saat} sp ${partie.spielerzahl} erw ${partie.erweiterung} comebackZuegeInPartie ${comebackZuege} zusatz ${JSON.stringify(vorher.zug.zusatzaktionen.map((z) => z.quelle))} comebackAktionenJetzt ${JSON.stringify(imZug)} ${aktion.typ}: vorsprung ${gewaehlt.vorsprung} statt ${kleinster.vorsprung}, wert ${gewaehlt.wert.toFixed(2)} gegen ${kleinster.wert.toFixed(2)}, vertieft ${vertieftGew.toFixed(2)} gegen ${vertieftKl.toFixed(2)} -> alt ${alt ? 'ok' : 'ROT'}`,
        )
      }
    }
    zeilen.push(`geprueft ${geprueft}, ausnahmen ${ausnahmenGesamt}, davon alt ok ${ausnahmenAlt}, nur vertieft ok ${nurVertieft}, grenze 20% ${Math.floor(0.2 * geprueft)}`)
    writeFileSync(process.env.X4_K3_ZIEL ?? 'x4-k3.txt', zeilen.join('\n') + '\n')
  }, 600_000)
})
