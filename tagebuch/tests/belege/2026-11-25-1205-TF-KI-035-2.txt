// tests/zz_pruefer_suche.test.ts (Probe des Prüfers K5, nach dem Lauf gelöscht)
// @vitest-environment node
import { it } from 'vitest'
import { anwendeAktion, ermittleAktionen, ermittleEntscheider, erstellePartie, erzeugeZufall } from '../src/engine'
import { baukasten } from '../src/engine/testing/szenario'
import type { Schlangenglied } from '../src/engine/testing/szenario'
import { sichtFuer, entscheiderVon } from '../src/ki/sicht'
import { planeZug } from '../src/ki/strategien/zugsuche'

function bunteSchlange(anzahl: number, versatz: number): Schlangenglied[] {
  const farben = ['Blau', 'Rot', 'Gelb', 'Violett', 'Braun'] as const
  return Array.from({ length: anzahl }, (_, stelle) => farben[(Math.floor(stelle / 2) + versatz) % farben.length])
}

it('Prüferprobe: Suchgrenzen', () => {
  const bau = baukasten({ spielerzahl: 3, erweiterung: false, zufall: erzeugeZufall(91) })
  for (const [spieler, versatz] of [[0, 0], [0, 1], [1, 2], [1, 3], [2, 4], [2, 0]] as const) bau.schlange(spieler, bunteSchlange(9, versatz))
  const zustand = bau.hand(1, ['Farbendieb', 'Schlangenfrass', 'Regenbogenschlange', 'Schlangenblockade', 'Grün']).haendeAuffuellen().amZug(1).zustand()
  const aktionen = ermittleAktionen(zustand)
  const spaet: string[] = []
  for (const saat of [1, 2, 3]) {
    const t = performance.now()
    const plan = planeZug(sichtFuer(zustand, 1), aktionen, erzeugeZufall(saat))
    spaet.push(`${(performance.now() - t).toFixed(1)} ms, suche ${plan.bewertungen.suche}`)
  }
  process.stderr.write(`PRUEFER Späte Stellung (${aktionen.length} Aktionen): ${spaet.join('; ')}\n`)

  let maxMs = 0
  let maxSuche = 0
  let schritte = 0
  for (const [spielerzahl, erweiterung, saat] of [[2, false, 424_242], [3, true, 515_151], [4, true, 616_161], [4, false, 717_171]] as const) {
    let z = erstellePartie({ spielerzahl, erweiterung, zufall: erzeugeZufall(saat) })
    const zufall = erzeugeZufall(saat + 1)
    for (let n = 0; n < 5000 && ermittleEntscheider(z) !== null; n++) {
      const e = entscheiderVon(z)
      if (e === null) break
      const a = ermittleAktionen(z)
      const t = performance.now()
      const plan = planeZug(sichtFuer(z, e.index), a, zufall)
      const ms = performance.now() - t
      maxMs = Math.max(maxMs, ms)
      maxSuche = Math.max(maxSuche, plan.bewertungen.suche)
      schritte++
      z = anwendeAktion(z, plan.aktion).zustand
    }
  }
  process.stderr.write(`PRUEFER Vier Partien solide KI gegen sich, ${schritte} Schritte: höchstens ${maxMs.toFixed(1)} ms, höchstens ${maxSuche} durchgerechnete Stellungen\n`)
}, 600_000)
